179    public static class EvaluateExpression {
180      /** Evaluate an expression */
181      public static double evaluateExpression(String expression) {
182        // Create operandStack to store operands
183        Stack<Double> operandStack = new Stack<Double>();
184  
185        // Create operatorStack to store operators
186        Stack<Character> operatorStack = new Stack<Character>();
187  
188        // Extract operands and operators
189        java.util.StringTokenizer tokens = new java.util.StringTokenizer(
190            expression, "()+-/*", true);
191  
192        LINECOMMENTPhase 1: Scan tokens
193        while (tokens.hasMoreTokens()) {
194          String token = tokens.nextToken().trim(); LINECOMMENTExtract a token
195          if (token.length() == 0) LINECOMMENTBlank space
196            continue; LINECOMMENTBack to the while loop to extract the next
197          LINECOMMENTtoken
198          else if (token.charAt(0) == '+' || token.charAt(0) == '-') {
199            LINECOMMENTProcess all +, -, *, / in the top of the operator stack
200            while (!operatorStack.isEmpty()
201                && (operatorStack.peek().equals('+')
202                    || operatorStack.peek().equals('-')
203                    || operatorStack.peek().equals('*') || operatorStack
204                    .peek().equals('/'))) {
205              processAnOperator(operandStack, operatorStack);
206            }
207  
208            LINECOMMENTPush the + or - operator into the operator stack
209            operatorStack.push(new Character(token.charAt(0)));
210          } else if (token.charAt(0) == '*' || token.charAt(0) == '/') {
211            LINECOMMENTProcess all *, / in the top of the operator stack
212            while (!operatorStack.isEmpty()
213                && (operatorStack.peek().equals('*') || operatorStack
214                    .peek().equals('/'))) {
215              processAnOperator(operandStack, operatorStack);
216            }
217  
218            LINECOMMENTPush the * or / operator into the operator stack
219            operatorStack.push(new Character(token.charAt(0)));
220          } else if (token.trim().charAt(0) == '(') {
221            operatorStack.push(new Character('(')); LINECOMMENTPush '(' to stack
222          } else if (token.trim().charAt(0) == ')') {
223            LINECOMMENTProcess all the operators in the stack until seeing '('
224            while (!operatorStack.peek().equals('(')) {
225              processAnOperator(operandStack, operatorStack);
226            }
227  
228            operatorStack.pop(); LINECOMMENTPop the '(' symbol from the stack
229          } else { LINECOMMENTAn operand scanned
230            LINECOMMENTPush an operand to the stack
231            operandStack.push(new Double(token));
232          }
233        }
234  
235        LINECOMMENTPhase 2: process all the remaining operators in the stack
236        while (!operatorStack.isEmpty()) {
237          processAnOperator(operandStack, operatorStack);
238        }
239  
240        LINECOMMENTReturn the result
241        return ((Double) (operandStack.pop())).doubleValue();
242      }

244      /**
245       * Process one opeator: Take an operator from operatorStack and apply it
246       * on the operands in the operandStack
247       */
248      public static void processAnOperator(Stack<Double> operandStack,
249          Stack<Character> operatorStack) {
250        if (operatorStack.peek().equals('+')) {
251          operatorStack.pop();
252          double op1 = ((Double) (operandStack.pop())).doubleValue();
253          double op2 = ((Double) (operandStack.pop())).doubleValue();
254          operandStack.push(new Double(op2 + op1));
255        } else if (operatorStack.peek().equals('-')) {
256          operatorStack.pop();
257          double op1 = ((Double) (operandStack.pop())).doubleValue();
258          double op2 = ((Double) (operandStack.pop())).doubleValue();
259          operandStack.push(new Double(op2 - op1));
260        } else if (operatorStack.peek().equals('*')) {
261          operatorStack.pop();
262          double op1 = ((Double) (operandStack.pop())).doubleValue();
263          double op2 = ((Double) (operandStack.pop())).doubleValue();
264          operandStack.push(new Double(op2 * op1));
265        } else if (operatorStack.peek().equals('/')) {
266          operatorStack.pop();
267          double op1 = ((Double) (operandStack.pop())).doubleValue();
268          double op2 = ((Double) (operandStack.pop())).doubleValue();
269          operandStack.push(new Double(op2 / op1));
270        }
271      }
