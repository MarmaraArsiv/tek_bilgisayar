 13  public class USMap extends Application {
 14    @Override // Override the start method in the Application class
 15    public void start(Stage primaryStage) {
 16      MyMap map = new MyMap();
 17      Scene scene = new Scene(map, 1200, 800);
 18      primaryStage.setTitle("USMap"); // Set the stage title
 19      primaryStage.setScene(scene); // Place the scene in the stage
 20      primaryStage.show(); // Display the stage
 21      
 22      map.setOnKeyPressed(e -> {
 23        if (e.getCode() == KeyCode.UP) {
 24          map.enlarge(); // Enlarge the map
 25        }
 26        else if (e.getCode() == KeyCode.DOWN) {
 27          map.shrink(); // SHrink the map
 28        }
 29      });
 30      map.requestFocus();
 31    }

 33    class MyMap extends BorderPane {
 34      private Group group = new Group();
 35      
 36      MyMap() {
 37        // Load coordinates from a file
 38        ArrayList<ArrayList<Point2D>> points = getPoints(); 
 39  
 40        // Add points to the polygon list
 41        for (int i = 0; i < points.size(); i++) {
 42          Polygon polygon = new Polygon();
 43          // Add points to the polygon list
 44          for (int j = 0; j < points.get(i).size(); j++)
 45            polygon.getPoints().addAll(points.get(i).get(j).getX(), 
 46              -points.get(i).get(j).getY());
 47          polygon.setFill(Color.WHITE);
 48          polygon.setStroke(Color.BLACK);
 49          polygon.setStrokeWidth(1 / 14.0);
 50          
 51          polygon.setOnMouseClicked(e -> {
 52            if (e.getButton() == MouseButton.PRIMARY) {
 53              polygon.setFill(Color.RED);
 54            }
 55            else if (e.getButton() == MouseButton.SECONDARY) {
 56              polygon.setFill(Color.BLUE);
 57            }
 58            else {
 59              polygon.setFill(Color.WHITE);
 60            }
 61          });
 62  
 63          group.getChildren().add(polygon);
 64        }
 65  
 66        group.setScaleX(14); 
 67        group.setScaleY(14);
 68        this.setCenter(group);
 69      }
 70      
 71      public void enlarge() {
 72        group.setScaleX(1.1 * group.getScaleX());
 73        group.setScaleY(1.1 * group.getScaleY());
 74      }
 75  
 76      public void shrink() {
 77        group.setScaleX(0.9 * group.getScaleX());
 78        group.setScaleY(0.9 * group.getScaleY());
 79      }
 80      
 81      private ArrayList<ArrayList<Point2D>> getPoints() {
 82        ArrayList<ArrayList<Point2D>> points = new ArrayList<>();
 83  
 84        try (Scanner input = new Scanner(new java.net.URL(
 85                "http://cs.armstrong.edu/liang/data/usmap.txt")
 86                .openStream())) {
 87          while (input.hasNext()) {
 88            String s = input.next(); 
 89            if (s.equals("USA")) { 
 90              int count = input.nextInt(); // Number of coordinates
 91              ArrayList<Point2D> state = new ArrayList<Point2D>();
 92              points.add(state); // Read a list of coordinates
 93              for (int i = 0; i < count; i++) {
 94                state.add(new Point2D(input.nextDouble(), input
 95                  .nextDouble()));
 96              }
 97            }
 98          }
 99        } 
100        catch (Exception ex) {
101          ex.printStackTrace();
102        }
103  
104        return points;
105      }
106    }
