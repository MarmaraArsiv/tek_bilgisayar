  1  import javafx.application.Application;
  2  import javafx.geometry.Pos;
  3  import javafx.scene.Scene;
  4  import javafx.scene.control.Button;
  5  import javafx.scene.layout.StackPane;
  6  import javafx.scene.layout.HBox;
  7  import javafx.scene.layout.BorderPane;
  8  import javafx.scene.paint.Color;
  9  import javafx.scene.shape.Circle;
 10  import javafx.stage.Stage;
 11  import javafx.scene.input.MouseButton;
 12  import javafx.scene.input.KeyCode;
 13  
 14  public class Test2 extends Application {
 15    @Override // Override the start method in the Application class
 16    public void start(Stage primaryStage) {
 17      StackPane pane = new StackPane();
 18      Circle circle = new Circle(50);
 19      circle.setStroke(Color.BLACK);
 20      circle.setFill(Color.WHITE);
 21      pane.getChildren().add(circle);
 22      
 23      HBox hBox = new HBox();
 24      hBox.setSpacing(10);
 25      hBox.setAlignment(Pos.CENTER);
 26      Button btEnlarge = new Button("Enlarge");
 27      
 28      btEnlarge.setOnAction(e -> {circle.setRadius(circle.getRadius() + 1);});
 29      Button btShrink = new Button("Shrink");
 30      btShrink.setOnAction(e -> {circle.setRadius(circle.getRadius() - 1);});
 31  
 32      circle.setOnMouseClicked(e -> {
 33        if (e.getButton() == MouseButton.PRIMARY)
 34          circle.setRadius(circle.getRadius() + 1);
 35        else if (e.getButton() == MouseButton.SECONDARY)
 36          circle.setRadius(circle.getRadius() - 1);
 37      });
 38      
 39      circle.setOnKeyPressed(e -> {
 40        if (e.getCode() == KeyCode.U) {
 41          circle.setRadius(circle.getRadius() + 1);
 42        }
 43        else if (e.getCode() == KeyCode.D) {
 44          circle.setRadius(circle.getRadius() - 1);
 45        }
 46      });
 47      
 48      hBox.getChildren().add(btEnlarge);
 49      hBox.getChildren().add(btShrink);
 50  
 51      BorderPane borderPane = new BorderPane();
 52      borderPane.setCenter(pane);
 53      borderPane.setBottom(hBox);
 54      BorderPane.setAlignment(hBox, Pos.CENTER);
 55      
 56      // Create a scene and place it in the stage
 57      Scene scene = new Scene(borderPane, 200, 150);
 58      primaryStage.setTitle("ControlCircle"); // Set the stage title
 59      primaryStage.setScene(scene); // Place the scene in the stage
 60      primaryStage.show(); // Display the stage
 61      
 62      circle.requestFocus();
 63    }
 64    
 65    /**
 66     * The main method is only needed for the IDE with limited
 67     * JavaFX support. Not needed for running from the command line.
 68     */
 69    public static void main(String[] args) {
 70      launch(args);
 71    }
 72  }
