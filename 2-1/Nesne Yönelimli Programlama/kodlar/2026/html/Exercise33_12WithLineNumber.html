186    public static class EvaluateExpression {
187      /** Evaluate an expression */
188      public static double evaluateExpression(String expression) {
189        // Create operandStack to store operands
190        Stack<Double> operandStack = new Stack<Double>();
191  
192        // Create operatorStack to store operators
193        Stack<Character> operatorStack = new Stack<Character>();
194  
195        // Extract operands and operators
196        java.util.StringTokenizer tokens = new java.util.StringTokenizer(
197            expression, "()+-/*", true);
198  
199        LINECOMMENTPhase 1: Scan tokens
200        while (tokens.hasMoreTokens()) {
201          String token = tokens.nextToken().trim(); LINECOMMENTExtract a token
202          if (token.length() == 0) LINECOMMENTBlank space
203            continue; LINECOMMENTBack to the while loop to extract the next
204          LINECOMMENTtoken
205          else if (token.charAt(0) == '+' || token.charAt(0) == '-') {
206            LINECOMMENTProcess all +, -, *, / in the top of the operator stack
207            while (!operatorStack.isEmpty()
208                && (operatorStack.peek().equals('+')
209                    || operatorStack.peek().equals('-')
210                    || operatorStack.peek().equals('*') || operatorStack
211                    .peek().equals('/'))) {
212              processAnOperator(operandStack, operatorStack);
213            }
214  
215            LINECOMMENTPush the + or - operator into the operator stack
216            operatorStack.push(new Character(token.charAt(0)));
217          } else if (token.charAt(0) == '*' || token.charAt(0) == '/') {
218            LINECOMMENTProcess all *, / in the top of the operator stack
219            while (!operatorStack.isEmpty()
220                && (operatorStack.peek().equals('*') || operatorStack
221                    .peek().equals('/'))) {
222              processAnOperator(operandStack, operatorStack);
223            }
224  
225            LINECOMMENTPush the * or / operator into the operator stack
226            operatorStack.push(new Character(token.charAt(0)));
227          } else if (token.trim().charAt(0) == '(') {
228            operatorStack.push(new Character('(')); LINECOMMENTPush '(' to stack
229          } else if (token.trim().charAt(0) == ')') {
230            LINECOMMENTProcess all the operators in the stack until seeing '('
231            while (!operatorStack.peek().equals('(')) {
232              processAnOperator(operandStack, operatorStack);
233            }
234  
235            operatorStack.pop(); LINECOMMENTPop the '(' symbol from the stack
236          } else { LINECOMMENTAn operand scanned
237            LINECOMMENTPush an operand to the stack
238            operandStack.push(new Double(token));
239          }
240        }
241  
242        LINECOMMENTPhase 2: process all the remaining operators in the stack
243        while (!operatorStack.isEmpty()) {
244          processAnOperator(operandStack, operatorStack);
245        }
246  
247        LINECOMMENTReturn the result
248        return ((Double) (operandStack.pop())).doubleValue();
249      }

251      /**
252       * Process one opeator: Take an operator from operatorStack and apply it
253       * on the operands in the operandStack
254       */
255      public static void processAnOperator(Stack<Double> operandStack,
256          Stack<Character> operatorStack) {
257        if (operatorStack.peek().equals('+')) {
258          operatorStack.pop();
259          double op1 = ((Double) (operandStack.pop())).doubleValue();
260          double op2 = ((Double) (operandStack.pop())).doubleValue();
261          operandStack.push(new Double(op2 + op1));
262        } else if (operatorStack.peek().equals('-')) {
263          operatorStack.pop();
264          double op1 = ((Double) (operandStack.pop())).doubleValue();
265          double op2 = ((Double) (operandStack.pop())).doubleValue();
266          operandStack.push(new Double(op2 - op1));
267        } else if (operatorStack.peek().equals('*')) {
268          operatorStack.pop();
269          double op1 = ((Double) (operandStack.pop())).doubleValue();
270          double op2 = ((Double) (operandStack.pop())).doubleValue();
271          operandStack.push(new Double(op2 * op1));
272        } else if (operatorStack.peek().equals('/')) {
273          operatorStack.pop();
274          double op1 = ((Double) (operandStack.pop())).doubleValue();
275          double op2 = ((Double) (operandStack.pop())).doubleValue();
276          operandStack.push(new Double(op2 / op1));
277        }
278      }
