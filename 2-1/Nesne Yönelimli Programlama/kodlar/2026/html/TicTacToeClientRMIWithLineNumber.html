 37    @Override // Override the start method in the Application class
 38    public void start(Stage primaryStage) {
 39      // Pane to hold cell
 40      GridPane pane = new GridPane(); 
 41      for (int i = 0; i < 3; i++)
 42        for (int j = 0; j < 3; j++)

 46      borderPane.setCenter(pane);

 48      borderPane.setBottom(lblIdentification);

 53      primaryStage.setScene(scene); // Place the scene in the stage
 54      primaryStage.show(); // Display the stage   
 55      
 56      new Thread( () -> {
 57      try {
 58        initializeRMI();
 59      }

 61        ex.printStackTrace();
 62      }}).start();

 65    /** Initialize RMI */
 66    protected boolean initializeRMI() throws Exception {
 67      String host = "";
 68  
 69      try {
 70        Registry registry = LocateRegistry.getRegistry(host);
 71        ticTacToe = (TicTacToeInterface) 
 72          registry.lookup("TicTacToeImpl");
 73        System.out.println
 74          ("Server object " + ticTacToe + " found");
 75      } 
 76      catch (Exception ex) {
 77        System.out.println(ex);
 78      }
 79  
 80      // Create callback for use by the 
 81      // server to control the client
 82      CallBackImpl callBackControl = new CallBackImpl(this);
 83  
 84      if (
 85        (marker = 
 86          ticTacToe.connect((CallBack)callBackControl)) != ' ')
 87      {
 88        System.out.println("connected as " + marker + " player.");
 89        lblIdentification.setText("You are player " + marker);
 90        return true;
 91      }
 92      else {
 93        System.out.println("already two players connected as ");
 94        return false;
 95      }
 96    }

114    public class Cell extends Pane {
115      // marked indicates whether the cell has been used

117

123  
124      public Cell(final int row, final int column) {
125        this.row = row;
126        this.column = column;
127        setStyle("-fx-border-color: black"); 
128        this.setPrefSize(2000, 2000);
129        this.setOnMouseClicked(e -> handleMouseClick());
130      }
131  
132      /** Return token */
133      public char getToken() {
134        return token;
135      }
136  
137      /** Set a new token */
138      public void setToken(char c) {
139        token = c;
140        marked = true;
141        
142        if (token == 'X') {
143          Line line1 = new Line(10, 10, 
144            this.getWidth() - 10, this.getHeight() - 10);
145          line1.endXProperty().bind(this.widthProperty().subtract(10));
146          line1.endYProperty().bind(this.heightProperty().subtract(10));
147          Line line2 = new Line(10, this.getHeight() - 10, 
148            this.getWidth() - 10, 10);
149          line2.startYProperty().bind(
150            this.heightProperty().subtract(10));
151          line2.endXProperty().bind(this.widthProperty().subtract(10));
152          
153          // Add the lines to the pane
154          this.getChildren().addAll(line1, line2); 
155        }
156        else if (token == 'O') {
157          Ellipse ellipse = new Ellipse(this.getWidth() / 2, 
158            this.getHeight() / 2, this.getWidth() / 2 - 10, 
159            this.getHeight() / 2 - 10);
160          ellipse.centerXProperty().bind(
161            this.widthProperty().divide(2));
162          ellipse.centerYProperty().bind(
163              this.heightProperty().divide(2));
164          ellipse.radiusXProperty().bind(
165              this.widthProperty().divide(2).subtract(10));        
166          ellipse.radiusYProperty().bind(
167              this.heightProperty().divide(2).subtract(10));   
168          ellipse.setStroke(Color.BLACK);
169          ellipse.setFill(Color.WHITE);
170          
171          getChildren().add(ellipse); // Add the ellipse to the pane
172        }
173      }
174  
175      /* Handle a mouse click event */
176      private void handleMouseClick() {
177        if (myTurn && !marked) {
178          // Mark the cell
179          setToken(marker);
180  
181          // Notify the server of the move
182          try {
183            ticTacToe.myMove(row, column, marker);
184          }
185          catch (RemoteException ex) {
186            System.out.println(ex);
187          }
188        }
189      }
190    }
