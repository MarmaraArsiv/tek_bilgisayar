#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <time.h>

void degerAta(int **, int, int);
void maxBul(int **, int, int, int[]);

int main()
{
	int satir,sutun;
	srand(time(NULL));
	
	printf("satir sayisi: ");
	scanf("%d",&satir);
	printf("sutun sayisi: ");
	scanf("%d",&sutun);
	
	int** array = (int**)malloc(satir * sizeof(int*));
    for (int i = 0; i < satir; i++)
        array[i] = (int*)malloc(sutun * sizeof(int));
        
	int maksimum[satir];
	degerAta(array,satir,sutun);	
	maxBul(array,satir,sutun,maksimum);
}

void degerAta(int **array, int satir, int sutun)
{
	for(int i=0; i<satir; i++)
	{
		for(int j=0; j<sutun; j++)
		{
			*(*(array+i)+j)=rand()%(90-20+1)+20;
			printf("%d ",*(*(array+i)+j)); //array[i][j]
		}
		printf("\n");
	}
	printf("\n");
}

void maxBul(int **array, int satir, int sutun, int maksimum[])
{
	int max;
	for(int i=0; i<satir; i++)
	{
		max=*(*(array+i));
		for(int j=0; j<sutun; j++)
		{
			if(*(*(array+i)+j)>max)
			{
				max=*(*(array+i)+j);
			}
		}
		maksimum[i]=max;
		printf("\n%d. satir max : %d ",i,max);
	}
}

void hisse_miktarlari_belirleme(hisse a[], char ad[]) {
    int i;
    a[0].yil = 2022;
    a[0].ay = 1;
    a[0].gun = 1;
    a[0].gunlukdeger = rand() % 500 + 500;
    strcpy(a[0].ad, ad);
    for (i = 1; i < sure; i++) {
        if (a[i - 1].gun == 29) {
            a[i - 1].gun = 1; 
            a[i - 1].ay += 1;
        }
        a[i].gunlukdeger = rand() % 1000 + 500 + rand() % 100 * 0.01; //gunluk degerler belirleniyor
        a[i].gun = a[i - 1].gun + 1;
        a[i].ay = a[i - 1].ay;
        a[i].yil = a[i - 1].yil;
        strcpy(a[i].ad, ad);
    }
}
void hisse_miktarlari_goster(hisse a[]) {
    int i;
    for (i = 0; i < sure; i++) {
        printf("%s:\t\t%.2f\t\t%d/%d/%d\n", a[i].ad, a[i].gunlukdeger, a[i].gun, a[i].ay, a[i].yil);
    }
}
void yatirim(hisse a[], int s) {
    int i, j, f, alis, satis, enbfark = 0;
    for (i = 0; i < s; i++) {
        for (j = i + 1; j < s; j++) {
            f = a[j].gunlukdeger - a[i].gunlukdeger;
            if (f > enbfark) {
                enbfark = f;
                alis = i;
                satis = j;
            }
        }
    }
    hisse_miktarlari_goster(a);
    printf("\n\n%s adli hisse %d/%d/%d tarihinde alinip %d/%d/%d tarihinde satilirsa en buyuk kar elde edilir.\nKar miktari : %.2f TL\nYatirim suresi: %d gun\n\n", 
    a[0].ad, a[alis].gun, a[alis].ay,a[alis].yil, a[satis].gun, a[satis].ay, 
    a[satis].yil,a[satis].gunlukdeger-a[alis].gunlukdeger,satis-alis);
}
